"""Diagnostic analysis of probe 2's own data (seed 44 regenerated deterministically; not a new probe)."""
import datetime as dt
import re
import statistics
import sys
from collections import defaultdict

from data.generator.config import BaselineIdentityConfig, GeneratorConfig
from data.generator.engine import generate_dataset
from data.generator.lpc5 import declaration as d
from data.generator.lpc5 import stats
from data.generator.lpc5.attributes import compute_tables
from data.generator.lpc5.availability import reference_availability
from data.generator.lpc5.frame import build_frame, knowledge
from data.generator.lpc5.judge import Allowlist, tally
from data.generator.population import build_universe

report = open(sys.argv[1], encoding="utf-8").read().splitlines()
pooled_cells = sorted(
    {
        (m.group(1), m.group(2), m.group(3))
        for line in report
        if (m := re.match(r"^      (?:R8|S1-U\(\w\)) (\w+)/([^/]+)/(.+)/POOLED: ", line))
    }
)
config = GeneratorConfig(
    seed=44, row_count=250_000, account_count=10_000, merchant_count=750, device_count=12_000,
    ip_count=5_000, fraud_rate=0.005, start_at=dt.datetime(2026, 1, 1, tzinfo=dt.UTC),
    end_at=dt.datetime(2026, 3, 1, tzinfo=dt.UTC), baseline_identity=BaselineIdentityConfig(),
)
universe = build_universe(config)
know = knowledge(universe)
frame = build_frame(generate_dataset(config, universe), seed=config.seed)
tables = compute_tables(frame, know, availability=reference_availability(know.start_ms))
allow = Allowlist()

print(f"== pooled cells failing R8 or S1-U: {len(pooled_cells)}")
summary = defaultdict(int)
for population_name, attribute, value in pooled_cells:
    population = d.Population(population_name)
    counts = tally(tables[population], attribute)
    for group, stratum, cell_value in counts.cells():
        if group != d.POOLED or cell_value != value:
            continue
        legit = counts.share(d.LEGIT, stratum, (value,))
        floor = max(legit.hi, d.LEGIT_SHARE_FLOOR)
        admitting, blocking = [], []
        for scenario, cell_stratum, v in counts.cells():
            if scenario in (d.LEGIT, d.POOLED) or cell_stratum != stratum or v != value:
                continue
            share = counts.share(scenario, stratum, (value,))
            if not share.r or share.x / share.r <= floor:
                continue
            tag = f"{scenario} {share.x}/{share.r}{' ENRICHED' if stats.enriched(share, legit) else ''}"
            if allow.is_ordinary(scenario, population, attribute, value):
                blocking.append(tag)
            else:
                admitting.append(tag)
        kind = "blocked-by-enriched" if any("ENRICHED" in b for b in blocking) else (
            "blocked-by-unenriched-only" if blocking else "no-blocker")
        summary[kind] += 1
        print(f"{population_name}/{attribute}/{value} [{stratum or '-'}] legit hi {legit.hi:.4f} :: {kind}")
        print(f"    blocking: {blocking}")
        print(f"    admitting: {admitting}")
print("== summary", dict(summary))

def per_instance(pattern):
    groups = defaultdict(list)
    for tx in frame.tx:
        if tx.group == pattern:
            groups[tx.cluster].append(tx)
    devices = [len({t.device for t in rows}) for rows in groups.values()]
    merchants = [len({t.merchant for t in rows}) for rows in groups.values()]
    countries = [len({t.country for t in rows}) for rows in groups.values()]
    homes = [len(know.home_devices.get(rows[0].account, ())) for rows in groups.values()]
    sizes = [len(rows) for rows in groups.values()]
    med = statistics.median
    print(f"== {pattern}: {len(groups)} instances; tx median {med(sizes)}; distinct devices median {med(devices)} max {max(devices)}; "
          f"home devices median {med(homes)}; distinct merchants median {med(merchants)}; distinct countries median {med(countries)} max {max(countries)}")

for pattern in ("VELOCITY_ATTACK", "CARD_TESTING"):
    per_instance(pattern)
