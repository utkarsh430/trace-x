"""Diagnostic: planted night share under M4 acceptance rules, against legitimate hours."""
import random
from data.generator.behavior import HOUR_WEIGHTS, WEEKDAY_WEIGHTS

H = 3_600_000
DAY = 24 * H
hour_total = sum(HOUR_WEIGHTS)
legit = [w / hour_total for w in HOUR_WEIGHTS]
legit_night = sum(legit[0:6])
max_h, max_d = max(HOUR_WEIGHTS), max(WEEKDAY_WEIGHTS)

def weight(ms):
    return (WEEKDAY_WEIGHTS[(ms // DAY + 3) % 7] / max_d) * (HOUR_WEIGHTS[(ms % DAY) // H] / max_h)

def ato(rng):
    k = rng.randint(1, 5)
    tx = sorted(rng.randrange(20 * 60_000, 6 * H) for _ in range(k))
    return [0], tx            # identity change at 0, transactions after

def travel(rng):
    distance = rng.uniform(1_000, 12_000)
    max_gap = distance / 900 * H
    gap = rng.uniform(0.15 * max_gap, 0.75 * max_gap)
    return [], [0, int(gap)]

def legit_hour(rng):
    target, running = rng.random() * hour_total, 0.0
    for h, w in enumerate(HOUR_WEIGHTS):
        running += w
        if target < running:
            return h
    return 23

def run(episode, rule, n=40_000, seed=1):
    rng = random.Random(seed)
    counts, kept = [0] * 24, 0
    while kept < n:
        others, tx = episode(rng)
        if rule == "anchor":
            base = rng.randrange(60) * DAY + legit_hour(rng) * H + rng.randrange(H) - (tx[0])
        else:
            base = rng.randrange(60 * DAY)
        times = [base + o for o in others + tx]
        if rule == "mean":
            accept = sum(weight(t) for t in times) / len(times)
        elif rule == "product":
            accept = 1.0
            for t in times:
                accept *= weight(t)
        else:
            accept = 1.0
        if rng.random() >= accept:
            continue
        kept += 1
        for o in tx:
            counts[((base + o) % DAY) // H] += 1
    total = sum(counts)
    shares = [c / total for c in counts]
    night = sum(shares[0:6])
    worst = max(s / l for s, l in zip(shares, legit))
    return night, worst

print(f"legitimate 00-05 share {legit_night:.4f}")
for name, episode in (("ACCOUNT_TAKEOVER", ato), ("IMPOSSIBLE_TRAVEL", travel)):
    for rule in ("mean", "product", "anchor"):
        night, worst = run(episode, rule)
        print(f"{name:<18} {rule:<8} planted 00-05 share {night:.4f} ({night / legit_night:.2f}x legit), worst hour ratio {worst:.2f}")
